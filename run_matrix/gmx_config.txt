# Optional structure curation with pdbfixer (receptors here are pre-curated)

fix_structure: no

# Fallback SEQRES source for structure curation, used only when a receptor

seqres_reference:

# Target production MD length in nanoseconds. Replaces manually editing
# `nsteps` in md.mdp.
#   - Complex belum punya hasil produksi -> dijalankan dari 0 ns sampai
#     production_ns (perilaku normal).
#   - Complex SUDAH punya md.cpt dari run sebelumnya (mis. selesai 1 ns) dan
#     production_ns lebih besar dari itu -> OTOMATIS dilanjutkan dari
#     checkpoint sampai production_ns (system-prep & equilibrasi di-skip,
#     hasil lama tidak dihapus/diulang).
#   - Complex sudah mencapai production_ns -> di-skip.
#   - Kosongkan value ini untuk kembali ke perilaku lama (pakai nsteps
#     bawaan md.mdp, selalu mulai dari 0 ns).
production_ns: 5

# Analisis pasca-produksi (analyze_md.py)
#
# Dibaca hanya oleh analyze_md.py; LAGMX.py mengabaikannya. Seluruh key di
# bawah ini opsional — config lama tanpa key ini tetap jalan dengan nilai
# bawaan yang sama seperti tertulis di sini.

#   cd run_matrix && python3 ../analyze_md.py

# Analisis yang dijalankan, dipisah koma. "all" menjalankan semuanya
# termasuk mmpbsa. Pilihan:
#   rmsd      stabilitas kompleks (backbone protein dan ligan terpisah)
#   rmsf      fleksibilitas per residu
#   rg        kompaktibilitas protein
#   sasa      luas permukaan protein, ligan, kompleks, dan area terkubur
#   hbond     ikatan hidrogen protein-ligan sepanjang lintasan
#   contacts  okupansi kontak per residu -> residu kunci
#   pca       gerakan global, proyeksi PC1/PC2
#   fel       free energy landscape di atas proyeksi PCA
#   mmpbsa    energi bebas ikatan lewat gmx_MMPBSA (butuh mmpbsa_python)
analysis: rmsd,rmsf,rg,sasa,hbond,contacts,pca,fel

# Berapa nanodetik pertama dibuang sebelum analisis. Awal produksi masih
# memuat relaksasi sisa dari ekuilibrasi; memasukkannya menggeser rerata RMSD
# dan mengaburkan plateau. Nol berarti seluruh lintasan dipakai.
analysis_skip_ns: 0

# Ambang kontak protein-ligan dalam nm. 0.4 nm adalah konvensi yang lazim
# dipakai untuk kontak non-ikatan.
analysis_contact_cutoff: 0.4

# gmx mana yang dipakai untuk analisis. Boleh berbeda dari gmx yang dipakai
# menjalankan simulasi — analisis tidak butuh GPU, jadi build apa pun cukup.
analysis_gmx: gmx

# gmx_MMPBSA hidup di environment terpisah karena menarik AmberTools versi
# sendiri. Isi dengan salah satu dari: prefix environment conda, interpreter
# python di dalamnya, atau path langsung ke executable gmx_MMPBSA. Dikosongkan
# berarti analisis mmpbsa dilewati, bahkan bila diminta di key `analysis`.
mmpbsa_python:

# gb, pb, atau both. MM/GBSA jauh lebih murah dan biasanya cukup untuk
# memeringkat ligan; MM/PBSA lebih mahal dan lebih tepat secara absolut.
mmpbsa_method: gb

# Dua cara menentukan frame mana yang disampling. `mmpbsa_interval` menang
# bila diisi.
#
# mmpbsa_interval: jarak antar frame, dalam satuan frame. Ini yang punya makna
#   fisik: frame yang berjarak lebih dekat daripada waktu autokorelasi sistem
#   menambah biaya tanpa menambah informasi, dan galat standar yang dihitung
#   atasnya tampak lebih kecil daripada yang sebenarnya karena rumusnya
#   mengasumsikan sampel independen.
#
#   Dengan nstxout-compressed 1000 dan dt 0.002 (satu frame tiap 2 ps),
#   pada produksi 100 ns dikurangi 10 ns awal = 45.000 frame tersisa:
#
#        interval    jarak      frame disampling
#              50    100 ps                 900
#             100    200 ps                 450
#             250    500 ps                 180   <- lazim dipakai
#             500    1,0 ns                  90
#
# mmpbsa_frames: dipakai hanya bila mmpbsa_interval kosong — jumlah frame yang
#   dituju, dan intervalnya dihitung dari panjang lintasan.
mmpbsa_interval: 250
mmpbsa_frames: 100

# Model Generalized Born (igb) dan konsentrasi garam dalam M.
mmpbsa_igb: 5
mmpbsa_salt: 0.150
